  localStream=await navigator.mediaDevices.getUserMedia({audio:true});
  voiceSocket=new WebSocket(WS_VOICE);

  voiceSocket.onopen=()=>{
   voiceSocket.send(JSON.stringify({
    action:"join",
    room:ROOM
   }));
   voiceStarted=true;
   micBtn.disabled=true;
   muteBtn.disabled=false;
   voiceStatus.textContent="🎙️ میکروفون روشن است";
  };

  voiceSocket.onmessage=async(event)=>{
   const data=JSON.parse(event.data);

   if(data.type==="voice_peer_joined"){
    const id="peer_"+Date.now()+"_"+Math.random();
    const pc=createVoicePeer(id);
    voicePeers[id]=pc;

    const offer=await pc.createOffer();
    await pc.setLocalDescription(offer);

    voiceSocket.send(JSON.stringify({
     action:"signal",
     data:{
      kind:"offer",
      from:id,
      sdp:pc.localDescription
     }
    }));
   }

   if(data.type==="signal"){
    await handleVoiceSignal(data.data);
   }

   if(data.type==="voice_peer_left"){
    for(const id in voicePeers){
     try{voicePeers[id].close()}catch(e){}
     delete voicePeers[id];
    }
   }
  };

  voiceSocket.onerror=()=>{
   voiceStatus.textContent="❌ اتصال صوتی برقرار نشد";
  };

 }catch(e){
  voiceStatus.textContent="❌ دسترسی به میکروفون داده نشد";
 }
}

function createVoicePeer(id){
 const pc=new RTCPeerConnection({
  iceServers:[{urls:"stun:stun.l.google.com:19302"}]
 });

 localStream.getTracks().forEach(track=>{
  pc.addTrack(track,localStream);
 });

 pc.onicecandidate=e=>{
  if(e.candidate && voiceSocket?.readyState===WebSocket.OPEN){
   voiceSocket.send(JSON.stringify({
    action:"signal",
    data:{
     kind:"ice",
     from:id,
     candidate:e.candidate
    }
   }));
  }
 };

 pc.ontrack=e=>{
  let audio=document.getElementById("audio_"+id);

  if(!audio){
   audio=document.createElement("audio");
   audio.id="audio_"+id;
   audio.autoplay=true;
   audio.controls=false;
   document.body.appendChild(audio);
  }

  audio.srcObject=e.streams[0];
 };

 return pc;
}

async function handleVoiceSignal(data){
 if(!data)return;

 if(data.kind==="offer"){
  const id=data.from;
  const pc=createVoicePeer(id);
  voicePeers[id]=pc;

  await pc.setRemoteDescription(data.sdp);

  const answer=await pc.createAnswer();
  await pc.setLocalDescription(answer);

  voiceSocket.send(JSON.stringify({
   action:"signal",
   data:{
    kind:"answer",
    from:id,
    sdp:pc.localDescription
   }
  }));
 }

 if(data.kind==="answer"){
  const pc=voicePeers[data.from];
  if(pc){
   await pc.setRemoteDescription(data.sdp);
  }
 }

 if(data.kind==="ice"){
  for(const id in voicePeers){
   try{
    await voicePeers[id].addIceCandidate(data.candidate);
   }catch(e){}
  }
 }
}

function stopVoice(){
 if(localStream){
  localStream.getTracks().forEach(t=>t.stop());
  localStream=null;
 }

 for(const id in voicePeers){
  try{voicePeers[id].close()}catch(e){}
 }

 voicePeers={};

 if(voiceSocket){
  try{voiceSocket.close()}catch(e){}
  voiceSocket=null;
 }

 voiceStarted=false;
 micBtn.disabled=false;
 muteBtn.disabled=true;
 voiceStatus.textContent="🔇 مکالمه صوتی خاموش است";
}

micBtn.onclick=startVoice;
muteBtn.onclick=stopVoice;
